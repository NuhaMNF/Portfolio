# Interactive 3D Box Field — Hero Background Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the hero's starfield with a full-hero interactive 3D scene: ~27 dark-violet rectangular box frames that light up violet/pink, rise, and tilt under the cursor, with a small warm glowing center, all on a pure black background.

**Architecture:** A lazy-loaded (dynamic import, `ssr: false`) React Three Fiber canvas renders 27 boxes generated by a deterministic seeded layout module. A window-level pointer listener writes NDC into a module store; each frame the scene raycasts it onto the structure's plane and writes a smoothed cursor point; each box damps its own influence and mutates refs only (zero React re-renders). Idle animation, reduced-motion, off-screen pause, and viewport-based scaling wrap it all.

**Tech Stack:** Next.js 16.3.1 (App Router), React 19.2.8, three, `@react-three/fiber` v9, `@react-three/drei` (only `useMediaQuery`), `@react-three/postprocessing` (Bloom), Tailwind v4.

**Spec:** `docs/superpowers/specs/2026-08-16-hero-boxfield-3d-design.md`

export type BoxSpec = {
  id: number;
  x: number;
  y: number;
  z: number;
  w: number;
  h: number;
  d: number;
  rotX: number;
  rotY: number;
  rotZ: number;
  phase: number;
  speed: number;
  depthFactor: number;
  centerBoost: number;
};

const BASE_W = 1.4;
const BASE_H = 0.9;
const BASE_D = 0.25;
const CELL_X = 1.3;
const CELL_Y = 1.1;
const ROW_COLS = [4, 6, 5, 6, 4] as const;
const BACK_LAYER_COUNT = 4;
const SKIP_CHANCE = 0.07;

function mulberry32(seed: number) {
  let a = seed >>> 0;
  return () => {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function clamp01(v: number) {
  return Math.min(1, Math.max(0, v));
}

export function generateBoxLayout(seed = 1337): BoxSpec[] {
  const rand = mulberry32(seed);
  const specs: BoxSpec[] = [];
  let id = 0;

  ROW_COLS.forEach((cols, row) => {
    const rowWidth = (cols - 1) * CELL_X;
    const offset = row % 2 === 1 ? CELL_X / 2 : 0;
    for (let col = 0; col < cols; col++) {
      if (rand() < SKIP_CHANCE) continue;
      const x = -rowWidth / 2 + col * CELL_X + offset + (rand() - 0.5) * 0.24;
      const y = -((ROW_COLS.length - 1) * CELL_Y) / 2 + row * CELL_Y + (rand() - 0.5) * 0.2;
      const back = rand() < 0.2;
      const z = (back ? -1.6 - rand() * 0.5 : 0) + (rand() - 0.5) * 0.3;
      specs.push({
        id: id++,
        x,
        y,
        z,
        w: BASE_W * (1 + (rand() - 0.5) * 0.16),
        h: BASE_H * (1 + (rand() - 0.5) * 0.16),
        d: BASE_D * (1 + (rand() - 0.5) * 0.16),
        rotX: (rand() - 0.5) * 0.08,
        rotY: (rand() - 0.5) * 0.08,
        rotZ: (rand() - 0.5) * 0.08,
        phase: rand() * Math.PI * 2,
        speed: 0.8 + rand() * 0.5,
        depthFactor: 0.7 + 0.3 * clamp01((z + 2) / 2.4),
        centerBoost: Math.max(0, 1 - Math.hypot(x, y) / 2.8) * 0.3,
      });
    }
  });

  for (let i = 0; i < BACK_LAYER_COUNT; i++) {
    const x = (rand() - 0.5) * 4.4;
    const y = (rand() - 0.5) * 2.6;
    const z = -1.9 - rand() * 0.5;
    specs.push({
      id: id++,
      x,
      y,
      z,
      w: BASE_W * (1 + (rand() - 0.5) * 0.16),
      h: BASE_H * (1 + (rand() - 0.5) * 0.16),
      d: BASE_D * (1 + (rand() - 0.5) * 0.16),
      rotX: (rand() - 0.5) * 0.1,
      rotY: (rand() - 0.5) * 0.1,
      rotZ: (rand() - 0.5) * 0.1,
      phase: rand() * Math.PI * 2,
      speed: 0.8 + rand() * 0.5,
      depthFactor: 0.62 + rand() * 0.08,
      centerBoost: 0,
    });
  }

  return specs;
}

"use client";

import { useEffect, useMemo, useRef, type RefObject } from "react";
import * as THREE from "three";
import { Canvas, useFrame } from "@react-three/fiber";
import { EffectComposer, Bloom } from "@react-three/postprocessing";
import { useReducedMotion } from "@/lib/hooks/useReducedMotion";
import { generateBoxLayout, type BoxSpec } from "@/components/ui/boxFieldLayout";

const REST_EDGE = new THREE.Color("#41306f");
const REST_EDGE_OPACITY = 0.32;
const REST_INTERIOR_OPACITY = 0.2;
const GROUP_ROT_X = -0.35;
const GROUP_ROT_Y = 0.47;
const GROUP_ROT_Z = -0.17;
const GROUP_X = 1.85;
const GROUP_Y = -0.3;

let canvasRect: DOMRect | null = null;

function makeGlowTexture() {
  const size = 128;
  const canvas = document.createElement("canvas");
  canvas.width = size;
  canvas.height = size;
  const g = canvas.getContext("2d");
  if (!g) return null;
  const grad = g.createRadialGradient(size / 2, size / 2, 0, size / 2, size / 2, size / 2);
  grad.addColorStop(0, "rgba(255, 233, 179, 0.9)");
  grad.addColorStop(0.25, "rgba(255, 215, 140, 0.35)");
  grad.addColorStop(1, "rgba(255, 200, 120, 0)");
  g.fillStyle = grad;
  g.fillRect(0, 0, size, size);
  const tex = new THREE.CanvasTexture(canvas);
  tex.colorSpace = THREE.SRGBColorSpace;
  return tex;
}

export function BoxFieldCanvas() {
  const containerRef = useRef<HTMLDivElement | null>(null);
  return (
    <div
      ref={containerRef}
      aria-hidden
      className="pointer-events-none absolute inset-0 z-0"
    >
      <Canvas
        dpr={[1, 1.75]}
        frameloop="demand"
        shadows
        gl={{ antialias: true, powerPreference: "high-performance" }}
        camera={{ fov: 40, position: [5, 4, 7], near: 0.1, far: 60 }}
        onCreated={({ camera, gl }) => {
          camera.lookAt(0, 0, 0);
          gl.setClearColor("#000000", 1);
        }}
      >
        <BoxFieldScene containerRef={containerRef} />
        <EffectComposer multisampling={0}>
          <Bloom
            mipmapBlur
            intensity={0.55}
            luminanceThreshold={0.35}
            luminanceSmoothing={0.45}
            radius={0.7}
          />
        </EffectComposer>
      </Canvas>
    </div>
  );
}

function BoxFieldScene({
  containerRef,
}: {
  containerRef: RefObject<HTMLDivElement | null>;
}) {
  const group = useRef<THREE.Group>(null);
  const specs = useMemo(() => generateBoxLayout(), []);
  const reduced = useReducedMotion();

  useEffect(() => {
    const el = containerRef.current;
    if (!el) return;
    const syncRect = () => {
      canvasRect = el.getBoundingClientRect();
    };
    syncRect();
    const ro = new ResizeObserver(syncRect);
    ro.observe(el);
    window.addEventListener("scroll", syncRect, { passive: true });
    window.addEventListener("resize", syncRect);
    return () => {
      ro.disconnect();
      window.removeEventListener("scroll", syncRect);
      window.removeEventListener("resize", syncRect);
    };
  }, [containerRef]);

  useFrame((state, dt) => {
    const g = group.current;
    if (!g) return;
    const t = state.clock.elapsedTime;
    if (reduced) {
      g.position.set(GROUP_X, GROUP_Y, 0);
      g.rotation.set(GROUP_ROT_X, GROUP_ROT_Y, GROUP_ROT_Z);
    } else {
      g.position.y = GROUP_Y + Math.sin((t * Math.PI * 2) / 7) * 0.05;
      g.rotation.z = GROUP_ROT_Z + Math.sin((t * Math.PI * 2) / 9) * 0.01;
      g.rotation.x = GROUP_ROT_X + Math.sin((t * Math.PI * 2) / 11) * 0.008;
    }
    state.invalidate();
    void dt;
  });

  return (
    <group
      ref={group}
      position={[GROUP_X, GROUP_Y, 0]}
      rotation={[GROUP_ROT_X, GROUP_ROT_Y, GROUP_ROT_Z]}
    >
      <ambientLight intensity={0.45} color="#56507e" />
      <directionalLight
        position={[4, 6, 3]}
        intensity={0.8}
        castShadow
        shadow-mapSize={[1024, 1024]}
        shadow-camera-left={-7}
        shadow-camera-right={7}
        shadow-camera-top={7}
        shadow-camera-bottom={-7}
        shadow-camera-near={0.5}
        shadow-camera-far={25}
      />
      {specs.map((s) => (
        <BoxModule key={s.id} spec={s} />
      ))}
      <CenterGlow />
    </group>
  );
}

function BoxModule({ spec }: { spec: BoxSpec }) {
  const geo = useMemo(() => new THREE.BoxGeometry(spec.w, spec.h, spec.d), [spec]);
  const edgesGeo = useMemo(() => new THREE.EdgesGeometry(geo), [geo]);
  const restOpacity = REST_EDGE_OPACITY * spec.depthFactor + spec.centerBoost * 0.4;
  const restInteriorOpacity =
    REST_INTERIOR_OPACITY * spec.depthFactor + spec.centerBoost * 0.25;

  return (
    <group position={[spec.x, spec.y, spec.z]} rotation={[spec.rotX, spec.rotY, spec.rotZ]}>
      <mesh geometry={geo} castShadow receiveShadow>
        <meshStandardMaterial
          color="#2a2050"
          emissive="#1a1038"
          emissiveIntensity={0.6}
          roughness={0.65}
          metalness={0.15}
          transparent
          opacity={restInteriorOpacity}
          depthWrite={false}
        />
      </mesh>
      <lineSegments geometry={edgesGeo}>
        <lineBasicMaterial
          color={REST_EDGE}
          transparent
          opacity={restOpacity}
          depthWrite={false}
          toneMapped={false}
        />
      </lineSegments>
    </group>
  );
}

function CenterGlow() {
  const tex = useMemo(makeGlowTexture, []);
  return (
    <group position={[0, 0, 0.35]}>
      <mesh>
        <sphereGeometry args={[0.2, 24, 24]} />
        <meshBasicMaterial color="#ffe9b3" toneMapped={false} />
      </mesh>
      {tex && (
        <sprite scale={[1.8, 1.8, 1]} renderOrder={2}>
          <spriteMaterial
            map={tex}
            transparent
            depthWrite={false}
            blending={THREE.AdditiveBlending}
            opacity={0.85}
            toneMapped={false}
          />
        </sprite>
      )}
      <pointLight color="#ffd27d" intensity={3} distance={7} decay={2} />
    </group>
  );
}

const MID_EDGE = new THREE.Color("#a855f7");
const HOT_EDGE = new THREE.Color("#ec4899");
const MAX_EDGE_OPACITY = 0.95;
const MAX_INTERIOR_OPACITY = 0.52;
const INFLUENCE_RADIUS_FULL = 0.6;
const INFLUENCE_RADIUS_MEDIUM = 1.4;
const INFLUENCE_RADIUS_WEAK = 2.6;
const LIFT = 0.09;
const SCALE_BOOST = 0.03;
const TILT_MAX = 0.06;
const DAMP_RATE = 8;
const CURSOR_DAMP_RATE = 6;

const pointerNdc = { x: 0, y: 0, active: false };
const cursorLocal = { x: 0, y: 0, z: 0, active: false };
let canvasRect: DOMRect | null = null;
let listenerInstalled = false;

function installPointerListener() {
  if (listenerInstalled || typeof window === "undefined") return;
  listenerInstalled = true;
  window.addEventListener("pointermove", (e) => {
    if (e.pointerType !== "mouse" || !canvasRect) return;
    pointerNdc.x = ((e.clientX - canvasRect.left) / canvasRect.width) * 2 - 1;
    pointerNdc.y = -((e.clientY - canvasRect.top) / canvasRect.height) * 2 + 1;
    pointerNdc.active = true;
  });
  window.addEventListener("pointerleave", () => {
    pointerNdc.active = false;
  });
  window.addEventListener("blur", () => {
    pointerNdc.active = false;
  });
}

function smoothstep(a: number, b: number, x: number) {
  const t = Math.min(1, Math.max(0, (x - a) / (b - a)));
  return t * t * (3 - 2 * t);
}

function computeInfluence(dist: number) {
  if (dist >= INFLUENCE_RADIUS_WEAK) return 0;
  if (dist < INFLUENCE_RADIUS_FULL) return 1;
  if (dist < INFLUENCE_RADIUS_MEDIUM) {
    return 1 - smoothstep(INFLUENCE_RADIUS_FULL, INFLUENCE_RADIUS_MEDIUM, dist) * 0.6;
  }
  return 0.4 * (1 - smoothstep(INFLUENCE_RADIUS_MEDIUM, INFLUENCE_RADIUS_WEAK, dist));
}

```tsx
function BoxModule({ spec, reduced }: { spec: BoxSpec; reduced: boolean }) {
  const group = useRef<THREE.Group>(null);
  const interiorMat = useRef<THREE.MeshStandardMaterial>(null);
  const edgeMat = useRef<THREE.LineBasicMaterial>(null);
  const influence = useRef(0);

  const geo = useMemo(() => new THREE.BoxGeometry(spec.w, spec.h, spec.d), [spec]);
  const edgesGeo = useMemo(() => new THREE.EdgesGeometry(geo), [geo]);
  const color = useMemo(() => REST_EDGE.clone(), []);

  const restOpacity = REST_EDGE_OPACITY * spec.depthFactor + spec.centerBoost * 0.4;
  const restInteriorOpacity =
    REST_INTERIOR_OPACITY * spec.depthFactor + spec.centerBoost * 0.25;

  useFrame((state, dt) => {
    const g = group.current;
    const im = interiorMat.current;
    const em = edgeMat.current;
    if (!g || !im || !em) return;
    const damp = 1 - Math.exp(-DAMP_RATE * dt);
    const target = cursorLocal.active
      ? computeInfluence(
          Math.hypot(spec.x - cursorLocal.x, spec.z - cursorLocal.z)
        )
      : 0;
    influence.current += (target - influence.current) * damp;
    const inf = influence.current;

    const moveMul = reduced ? 0.2 : 1;

    g.position.x = spec.x;
    g.position.y = spec.y + inf * LIFT * moveMul;
    g.position.z = spec.z;

    const s = 1 + inf * SCALE_BOOST * moveMul;
    g.scale.setScalar(s);

    const dirX = cursorLocal.active ? spec.x - cursorLocal.x : 0;
    const dirZ = cursorLocal.active ? spec.z - cursorLocal.z : 0;
    const tiltX = THREE.MathUtils.clamp(dirZ * 0.9, -1, 1) * TILT_MAX * inf * moveMul;
    const tiltY = THREE.MathUtils.clamp(dirX * 0.9, -1, 1) * TILT_MAX * 0.6 * inf * moveMul;
    g.rotation.set(spec.rotX + tiltX, spec.rotY + tiltY, spec.rotZ);

    if (inf <= 0.5) color.copy(REST_EDGE).lerp(MID_EDGE, inf * 2);
    else color.copy(MID_EDGE).lerp(HOT_EDGE, (inf - 0.5) * 2);
    color.multiplyScalar(0.82 + 0.18 * spec.depthFactor + spec.centerBoost * 0.55);

    em.color.copy(color);
    em.opacity = Math.min(MAX_EDGE_OPACITY, restOpacity + inf * 0.62);
    im.opacity = Math.min(MAX_INTERIOR_OPACITY, restInteriorOpacity + inf * 0.3);
  });

  return (
    <group
      ref={group}
      position={[spec.x, spec.y, spec.z]}
      rotation={[spec.rotX, spec.rotY, spec.rotZ]}
    >
      <mesh geometry={geo} castShadow receiveShadow>
        <meshStandardMaterial
          ref={interiorMat}
          color="#2a2050"
          emissive="#1a1038"
          emissiveIntensity={0.6}
          roughness={0.65}
          metalness={0.15}
          transparent
          opacity={restInteriorOpacity}
          depthWrite={false}
        />
      </mesh>
      <lineSegments geometry={edgesGeo}>
        <lineBasicMaterial
          ref={edgeMat}
          color={REST_EDGE}
          transparent
          opacity={restOpacity}
          depthWrite={false}
          toneMapped={false}
        />
      </lineSegments>
    </group>
  );
}
```
